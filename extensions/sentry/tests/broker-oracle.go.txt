    "context"
    "encoding/base64"
    "encoding/json"
    "io"
    "net"
    "net/http"
    "net/http/httptest"

    "github.com/nwverse/persona/sdk/go/wire"
)

// Synthetic broker policy fixture only, not an installed hub or real Sentry endpoint.
func TestSentryOriginalWildcardBrokerFixture(t *testing.T) {
    manifest, err := os.ReadFile("sentry-source-manifest.json")
    if err != nil { t.Fatal(err) }
    if err := wire.ValidateJSON("Manifest", manifest); err != nil { t.Fatal("actual source manifest validation", err) }
    var m struct { Permissions struct { Network []string } }
    if err := json.Unmarshal(manifest, &m); err != nil { t.Fatal(err) }
    if len(m.Permissions.Network) != 1 || m.Permissions.Network[0] != "*.sentry.io" { t.Fatal("original grant changed") }
    var hits int
    endpoint := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        hits++
        body, _ := io.ReadAll(r.Body)
        if r.Method != "POST" || r.URL.Path != "/api/42/envelope/" || string(body) != "synthetic-envelope\n" || r.Header.Get("X-Sentry-Auth") != "Sentry sentry_key=fixture-key" { t.Error("synthetic envelope altered") }
        io.WriteString(w, "synthetic-receipt")
    }))
    defer endpoint.Close()
    p := &Proxy{
        Upstream: func(*url.URL) (*url.URL, error) { return nil, nil },
        LocalAddrs: func() []net.IP { return nil },
        Resolve: func(context.Context, string) ([]net.IP, error) { return []net.IP{net.ParseIP("192.0.2.1")}, nil },
        Dial: func(ctx context.Context, _, _ string) (net.Conn, error) { return (&net.Dialer{}).DialContext(ctx, "tcp", strings.TrimPrefix(endpoint.URL, "http://")) },
    }
    defer p.Close()
    p.Register("sentry", Rule{Hosts:m.Permissions.Network})
    request := func(host string) int {
        raw, _ := json.Marshal(map[string]any{"url":"http://"+host+"/api/42/envelope/", "method":"POST", "headers":map[string]string{"X-Sentry-Auth":"Sentry sentry_key=fixture-key"}, "body":base64.StdEncoding.EncodeToString([]byte("synthetic-envelope\n"))})
        w := httptest.NewRecorder()
        p.handler("sentry").ServeHTTP(w, httptest.NewRequest("POST", RuntimeFetchPath, bytes.NewReader(raw)))
        return w.Code
    }
    if code := request("o1.ingest.sentry.io"); code != 200 { t.Fatalf("original wildcard unsupported by actual broker: %d", code) }
    for _, host := range []string{"sentry.io", "sentry.io.example.com", "errors.example.com"} {
        if code := request(host); code != 403 { t.Fatalf("undeclared host accepted: %d", code) }
    }
    if hits != 1 { t.Fatal("denied request reached synthetic endpoint") }
    t.Log("PASS actual source-manifest validator and runtime-fetch broker: original *.sentry.io; one accepted synthetic receipt; three undeclared hosts denied; no added grants")
}
