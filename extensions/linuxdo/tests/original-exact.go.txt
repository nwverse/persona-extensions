package main

import (
    "context"
    "encoding/json"
    "fmt"
    "os"
    "testing"
)

// These controls exercise the unchanged Go tool handler, not a replacement decoder.
func TestExactOriginalControls(t *testing.T) {
    data, err := os.ReadFile("tests/exact-tool-cases.json")
    if err != nil { t.Fatal(err) }
    var cases []struct {
        Args string `json:"args"`
        OK bool `json:"ok"`
    }
    if err := json.Unmarshal(data, &cases); err != nil { t.Fatal(err) }
    f := newFakeSite(t)
    f.topics[1] = topicJSON(1, "Synthetic", "Opening")
    p := testPlugin(t, f, nil)
    for index, tc := range cases {
        raw := json.RawMessage(`{"name":"linuxdo_topic","args":` + tc.Args + `}`)
        _, err := p.cToolsCall(context.Background(), nil, raw)
        if (err == nil) != tc.OK { t.Fatalf("case %d: success=%v want=%v", index, err == nil, tc.OK) }
        fmt.Printf("original exact tool control: case=%d success=%v\n", index, err == nil)
    }
}
